Syntax

Prefer compact steps.

steps:
  - goto: $base_url
  - fill: "#email"
    with: $user
  - click: "#login-btn"
  - assert: ".welcome"
    text: "Hello"

POM

imports:
  - pages/login.yaml
steps:
  - do: login.login
    user: $user
    pass: $pass

Cases

cases:
  - { name: a, user: a@test.com, expect: "Welcome A" }
steps:
  - fill: "#email"
    with: $user
  - assert: ".welcome"
    text: $expect

HTTP

engine: http
steps:
  - get: $api_base/get
    save: ping
  - assert: $ping.status == 200
